cmake_minimum_required(VERSION 3.16)
project(ppsspp_netpacket_tests LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(NetpacketAdhocTest NetpacketAdhocTest.cpp ../NetpacketAdhoc.cpp)
target_compile_features(NetpacketAdhocTest PRIVATE cxx_std_17)
target_include_directories(NetpacketAdhocTest PRIVATE ../..)
target_link_libraries(NetpacketAdhocTest PRIVATE Threads::Threads)
add_executable(NetpacketCoreSmoke NetpacketCoreSmoke.cpp ../NetpacketAdhoc.cpp)
target_compile_features(NetpacketCoreSmoke PRIVATE cxx_std_17)
target_include_directories(NetpacketCoreSmoke PRIVATE ../.. ../libretro-common/include)
target_link_libraries(NetpacketCoreSmoke PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
	target_link_libraries(NetpacketCoreSmoke PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME NetpacketAdhoc COMMAND NetpacketAdhocTest)
